# PC → PLC → 로봇 픽앤플레이스 E2E (spec docs/superpowers/specs/2026-11-06-plc-e2e-ladder-design.md)
# 한 줄 = 렁 하나. 형식: 동작[, 동작…] := 조건 ; 설명

# 기호 → 디바이스. 그리퍼가 오면 GRIP/RELEASE 두 줄만 P 주소로 바꾼다.
MAP DI0 P00044 ; 로봇 DI0 움직임1 요청
MAP DI1 P00045 ; 로봇 DI1 움직임2 요청
MAP DO0 P00004 ; 로봇 DO0 움직임1 종료
MAP DO1 P00005 ; 로봇 DO1 움직임2 종료
MAP GRIP M00320 ; 그리퍼 잡기 (그리퍼 도착 전 임시)
MAP RELEASE M00321 ; 그리퍼 놓기 (그리퍼 도착 전 임시)

BRST M00310 5, SET M00310, MOV 0 D01000 := F0009B ; E2E 픽앤플레이스(PC 1111 → 움직임1 → 잡기 → 움직임2 → 놓기). 첫 스캔: 대기 단계, D01000=0
OUT M00300 := [= D01000 1111] ; 시작 명령: PC가 D01000에 1111
SET M00311, RST M00310 := M00300 & M00310 ; 대기 → 움직임1
SET M00312, RST M00311, MOV 2222 D01000 := ^DO0 & M00311 ; 움직임1 종료(DO0) → 잡기, 2222
SET M00313, RST M00312, MOV 4444 D01000 := T0200 & M00312 ; 잡기 2초 끝 → 움직임2, 4444
SET M00314, RST M00313, MOV 3333 D01000 := ^DO1 & M00313 ; 움직임2 종료(DO1) → 놓기, 3333
SET M00310, RST M00314, MOV 5555 D01000 := T0201 & M00314 ; 놓기 2초 끝 → 대기, 5555
TON T0200 20 := M00312 ; 잡기 2초 (100ms x 20)
TON T0201 20 := M00314 ; 놓기 2초 (100ms x 20)
OUT DI0 := M00311 ; 로봇 DI0 움직임1 요청
OUT DI1 := M00313 ; 로봇 DI1 움직임2 요청
OUT GRIP := M00312 & !RELEASE ; 그리퍼 잡기 (놓기와 인터록)
OUT RELEASE := M00314 & !GRIP ; 그리퍼 놓기 (잡기와 인터록)
